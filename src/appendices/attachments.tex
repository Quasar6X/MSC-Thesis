<constraint>               ::= <suite> "&" <build-option-constraints>

<build-option-constraints> ::= <build-option-constraints> "&" <build-option>
                             | <build_option>

<build-option>             ::= <id>
                             | <id> ":" <id>

<suite>                    ::= "*"
                             | <id>

<id>                       ::= non empty string
    \end{minted}

    def get_item_default(self, item: str, default: str | None = None) -> str: ...
    def get_bool(self, item: str, *, default: bool | None = None) -> bool: ...

    def get_int(self, item: str, default: int | None = None) -> int: ...
    def get_int_default(self, item: str, default: int | None = None) -> int: ...
    def get_float(self, item: str, default: float | None = None) -> float: ...
    def get_float_default(self, item: str, default: float | None = None) -> float: ...
    def get_lines(self, item: str, default: str | None = None) -> list[str]: ...
    def get_lines_default(self, item: str, default: str | None = None) -> list[str]: ...
    def get_list(self, item: str, default: str | None = None, sep: str | None = None) -> list[str]: ...
    def get_list_default(self, item: str, default: str | None = None, sep: str | None = None) -> list[str]: ...
