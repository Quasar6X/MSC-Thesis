\section{Motiváció}

    \begin{frame}{Motiváció}{Különbségek alternatív megoldásokhoz képest}
        Természetesen a \alert{SuperSet} előtt is léteztek megoldások:
        \begin{tasks} (2)
            \task{} \textit{GCC DejaGNU}
            \task{} \textit{LLVM Lit}
        \end{tasks}

        A \alert{SuperSet} azonban többet kínál ezen megoldásoknál:
        \begin{tasks} (2)
            \task{} Bővíthető \textit{DSL} konfiguráció
            \task{} Bővítmény rendszer
            \task{} Caching stratégia
            \task{} Report generálás
            \task{} Elosztott működés
            \task{} Teszt generálás
        \end{tasks}
    \end{frame}

    \begin{frame}{Motiváció}{Erőforrás kihasználás}
        A \alert{SuperSet} fő motivációja az erőforrások jobb kihasználása.

        \vspace{12pt}
        \begin{columns}[T]
            \begin{column}{.475\textwidth}
                A probléma:

                \begin{itemize}
                    \item Rengeteg teszt eset \textit{\~{}250\,000}
                    \item Még több kapcsoló kombináció:
                          \begin{tasks} (2)
                              \task{} Target
                              \task{} OptLevel
                              \task{} ABI
                              \task{} \dots
                          \end{tasks}
                    \item \dots exponenciális robbanás
                \end{itemize}
            \end{column}

            \begin{column}{.475\textwidth}
                Futásidőt redukáló tényezők:
                \begin{itemize}
                    \item Teszcsomagok közötti átfedés
                    \item Fordítási idejű tesztek (nincs szimulátor)
                \end{itemize}
            \end{column}
        \end{columns}
    \end{frame}

\section{Implementáció}

    \begin{frame}{Implementáció}{Standard és bővíthető konfiguráció}
        \begin{itemize}
            \item Mappa struktúra független
            \item Hierarchikus konfigurálhatóság az \textit{.INI szekciók} által
            \item Teszt redukciós beállítások: \textit{MATRIX, CONSTRAINTS}
            \item Lehetőség saját kapcsolók definiálására \textit{(DSL)}
        \end{itemize}
    \end{frame}

    \begin{frame}{Implementáció}{Tesztgenerálás}
        \begin{itemize}
            \item Lehetőség tesztgeneráló alkalmazások integrálására
                  \begin{itemize}
                      \item Jelenleg: \textit{Csmith}
                  \end{itemize}
            \item A tesztesetek formátuma átalakításra kerül a \alert{SuperSet} formátumra
            \item Teszteset validálás host fordítókkal
        \end{itemize}
    \end{frame}

    \begin{frame}{Implementáció}{Bővítmény rendszer}
        A bővítmény rendszer lehetőséget ad a tesztcsomagok futtatási parancsainak személyreszabására, különböző
        fordítóprogramok számára.

        \vspace{12pt}
        \begin{itemize}
            \item Teszt életciklus kezelés
            \item Cache stratégia meghatározás
            \item Futás idejű betöltés
        \end{itemize}
    \end{frame}

    \begin{frame}{Implementáció}{Futtatás és eredmény reportálás}
        \begin{itemize}
            \item Lehetőség lokális futtatásra
            \item Eredmény tárolás \textit{SQLite} adatbázisban
            \item Elosztott futtatás
                  \begin{itemize}
                      \item Kommunikáció \textit{WebSocketen} át
                      \item Saját állapotalapú protokoll
                      \item Feladat felosztási heurisztika a kliensek képességei alapján
                  \end{itemize}
            \item Eredmény jelentés készítés \textit{(HTML, JUnit XML)}
        \end{itemize}
    \end{frame}
